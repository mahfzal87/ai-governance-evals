# Release Notes → Blog Post Publisher

Below is a complete, production-ready tool (`release-notes-publisher`) that:

1. **Pulls** internal release notes (from `CHANGELOG.md`, a GitHub Release, or a plain file)
2. **Transforms** them into a polished blog post (LLM-based, with a deterministic no-API-key fallback)
3. **Publishes** to your blog (Ghost, WordPress, or a git-based static site), or does a safe `--dry-run`
4. **Tracks state** so re-running the same release never double-publishes
5. Ships with **tests**, a **GitHub Actions workflow**, and a **Dockerfile**

### Design decisions / assumptions (adjust as needed)

- Defaults to **draft** status everywhere — you must pass `--publish` to go live. This prevents an LLM hallucination or bad changelog entry from going straight to production.
- Idempotency is handled via a small JSON state file keyed by version/tag — safe to re-run in CI.
- Pluggable architecture (`Source` / `Transformer` / `Publisher` interfaces) so you can add Confluence, Notion, Webflow, Contentful, etc. without touching the core.
- Secrets are pulled from environment variables, referenced in YAML config via `${VAR_NAME}` — nothing sensitive is ever committed.

## `release_notes_publisher/config.py`

```python
import os
import re
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Dict

import yaml

from .exceptions import ConfigError

ENV_PATTERN = re.compile(r"\$\{([A-Z0-9_]+)\}")


def _resolve_env(value: Any) -> Any:
    """Recursively substitute ${VAR_NAME} references with environment variables."""
    if isinstance(value, str):
        def replace(match: "re.Match[str]") -> str:
            var = match.group(1)
            if var not in os.environ:
                raise ConfigError(
                    f"Environment variable '{var}' referenced in config is not set"
                )
            return os.environ[var]

        return ENV_PATTERN.sub(replace, value)
    if isinstance(value, dict):
        return {k: _resolve_env(v) for k, v in value.items()}
    if isinstance(value, list):
        return [_resolve_env(v) for v in value]
    return value


@dataclass
class AppConfig:
    raw: Dict[str, Any]

    @classmethod
    def load(cls, path: str) -> "AppConfig":
        p = Path(path)
        if not p.exists():
            raise ConfigError(f"Config file not found: {path}")
        with p.open("r", encoding="utf-8") as f:
            data = yaml.safe_load(f) or {}
        # Only resolve env vars lazily section-by-section would be nicer for
        # partial failures, but resolving eagerly keeps behavior predictable.
        try:
            data = _resolve_env(data)
        except ConfigError:
            # Re-raise as-is; env resolution errors are already descriptive.
            raise
        return cls(raw=data)

    def section(self, name: str) -> Dict[str, Any]:
        return self.raw.get(name, {}) or {}
```

## `release_notes_publisher/state.py`

```python
import json
import os
import tempfile
from pathlib import Path
from typing import Any, Dict


class StateStore:
    """Tracks which release versions have already been published.

    Backed by a simple JSON file. Writes are atomic (write to a temp file,
    then os.replace) so a killed process can't corrupt the file. This is
    sufficient for the common case of a single CI job running at a time.
    If you need concurrency-safe writes across parallel jobs, swap this for
    a database or a locking remote store (e.g. S3 with conditional writes).
    """

    def __init__(self, path: str):
        self.path = Path(path)
        if not self.path.exists():
            self.path.parent.mkdir(parents=True, exist_ok=True)
            self._write({})

    def _read(self) -> Dict[str, Any]:
        if not self.path.exists():
            return {}
        with self.path.open("r", encoding="utf-8") as f:
            content = f.read().strip()
            return json.loads(content) if content else {}

    def _write(self, data: Dict[str, Any]) -> None:
        parent = str(self.path.parent) or "."
        fd, tmp_path = tempfile.mkstemp(dir=parent, prefix=".state_", suffix=".tmp")
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as f:
                json.dump(data, f, indent=2, sort_keys=True)
            os.replace(tmp_path, self.path)
        finally:
            if os.path.exists(tmp_path):
                os.remove(tmp_path)

    def is_published(self, ref: str) -> bool:
        return ref in self._read()

    def get(self, ref: str) -> Dict[str, Any]:
        return self._read().get(ref, {})

    def mark_published(self, ref: str, info: Dict[str, Any]) -> None:
        data = self._read()
        data[ref] = info
        self._write(data)

    def all(self) -> Dict[str, Any]:
        return self._read()
```

## `release_notes_publisher/http_utils.py`

```python
import logging

import requests
from tenacity import (
    before_sleep_log,
    retry,
    retry_if_exception,
    stop_after_attempt,
    wait_exponential,
)

logger = logging.getLogger(__name__)


def _is_retryable(exc: BaseException) -> bool:
    if isinstance(exc, (requests.exceptions.ConnectionError, requests.exceptions.Timeout)):
        return True
    if isinstance(exc, requests.exceptions.HTTPError):
        response = exc.response
        return response is not None and response.status_code >= 500
    return False


def request_with_retry(method: str, url: str, **kwargs: object) -> requests.Response:
    """A thin wrapper around requests.request with retries on transient failures.

    Retries connection errors, timeouts, and 5xx responses with exponential
    backoff. 4xx responses are returned as-is so callers can handle them
    (e.g. surface a clear "invalid config" error instead of retrying forever).
    """

    @retry(
        reraise=True,
        stop=stop_after_attempt(4),
        wait=wait_exponential(multiplier=1, min=1, max=10),
        retry=retry_if_exception(_is_retryable),
        before_sleep=before_sleep_log(logger, logging.WARNING),
    )
    def _do_request() -> requests.Response:
        resp = requests.request(method, url, **kwargs)
        if resp.status_code >= 500:
            resp.raise_for_status()
        return resp

    return _do_request()
```

### `release_notes_publisher/sources/git_tag.py`

```python
import re
from datetime import datetime
from pathlib import Path

from ..exceptions import SourceError
from ..models import ReleaseNotes
from .base import Source

# Matches "Keep a Changelog"-style headers, e.g.:
#   ## [1.2.0] - 2024-06-01
#   ## v1.2.0
#   ## 1.2.0 - 2024-06-01
VERSION_HEADER_RE = re.compile(
    r"^##\s*\[?v?(?P<version>[\w.\-]+)\]?\s*(?:-\s*(?P<date>\d{4}-\d{2}-\d{2}))?\s*$",
    re.MULTILINE,
)


class GitTagChangelogSource(Source):
    """Extracts the section for a given version out of a CHANGELOG.md file."""

    def __init__(self, changelog_path: str = "CHANGELOG.md"):
        self.changelog_path = Path(changelog_path)

    def fetch(self, ref: str) -> ReleaseNotes:
        if not self.changelog_path.exists():
            raise SourceError(f"Changelog file not found: {self.changelog_path}")

        content = self.changelog_path.read_text(encoding="utf-8")
        normalized_ref = ref.lstrip("vV")

        matches = list(VERSION_HEADER_RE.finditer(content))
        if not matches:
            raise SourceError(f"No version headers found in {self.changelog_path}")

        for i, m in enumerate(matches):
            version = m.group("version")
            if version in (ref, normalized_ref):
                start = m.end()
                end = matches[i + 1].start() if i + 1 < len(matches) else len(content)
                section = content[start:end].strip()
                if not section:
                    raise SourceError(f"Changelog section for {ref} is empty")
                date_str = m.group("date")
                release_date = (
                    datetime.strptime(date_str, "%Y-%m-%d") if date_str else None
                )
                return ReleaseNotes(version=ref, raw_markdown=section, release_date=release_date)

        raise SourceError(f"Version '{ref}' not found in {self.changelog_path}")
```

### `release_notes_publisher/sources/github_release.py`

```python
import os
from datetime import datetime

import requests

from ..exceptions import SourceError
from ..http_utils import request_with_retry
from ..models import ReleaseNotes
from .base import Source


class GitHubReleaseSource(Source):
    """Fetches release notes from a GitHub Release's body."""

    def __init__(
        self,
        repo: str,
        token_env: str = "GITHUB_TOKEN",
        api_base: str = "https://api.github.com",
    ):
        self.repo = repo
        self.token = os.environ.get(token_env)
        self.api_base = api_base.rstrip("/")

    def fetch(self, ref: str) -> ReleaseNotes:
        url = f"{self.api_base}/repos/{self.repo}/releases/tags/{ref}"
        headers = {"Accept": "application/vnd.github+json"}
        if self.token:
            headers["Authorization"] = f"Bearer {self.token}"

        try:
            resp = request_with_retry("GET", url, headers=headers, timeout=15)
        except requests.exceptions.RequestException as e:
            raise SourceError(f"GitHub API request failed: {e}") from e

        if resp.status_code == 404:
            raise SourceError(f"GitHub release not found for tag '{ref}' in {self.repo}")
        if resp.status_code >= 400:
            raise SourceError(f"GitHub API error ({resp.status_code}): {resp.text}")

        data = resp.json()
        body = (data.get("body") or "").strip()
        if not body:
            raise SourceError(f"GitHub release '{ref}' has an empty body")

        published_at = data.get("published_at")
        release_date = (
            datetime.fromisoformat(published_at.replace("Z", "+00:00"))
            if published_at
            else None
        )
        return ReleaseNotes(
            version=ref,
            raw_markdown=body,
            release_date=release_date,
            source_url=data.get("html_url"),
        )
```

### `release_notes_publisher/transformers/template_transformer.py`

```python
import re
from typing import List, Optional

from ..models import BlogPost, ReleaseNotes
from .base import Transformer


def slugify(text: str) -> str:
    text = text.lower().strip()
    text = re.sub(r"[^a-z0-9]+", "-", text)
    return re.sub(r"-{2,}", "-", text).strip("-")


class TemplateTransformer(Transformer):
    """Deterministic, dependency-free transformer.

    Wraps the raw release notes with a light narrative shell. This is the
    safe default when no LLM API key is configured, and it's what the test
    suite uses since it has no external dependencies or network calls.
    """

    def __init__(self, default_tags: Optional[List[str]] = None):
        self.default_tags = default_tags or ["release-notes"]

    def transform(self, notes: ReleaseNotes) -> BlogPost:
        title = f"What's new in {notes.version}"
        intro = (
            f"We've just shipped {notes.version}. Here's a rundown of what's new, "
            "improved, and fixed in this release.\n\n"
        )
        body = intro + notes.raw_markdown.strip()

        first_line = next(
            (line.strip("-* ").strip() for line in notes.raw_markdown.splitlines() if line.strip()),
            title,
        )
        excerpt = first_line[:200]

        return BlogPost(
            title=title,
            slug=slugify(f"{notes.version}-release-notes"),
            excerpt=excerpt,
            body_markdown=body,
            tags=list(self.default_tags),
            version=notes.version,
        )
```

### `release_notes_publisher/transformers/llm_transformer.py`

```python
import json
import logging
import os
from typing import List, Optional

from ..exceptions import ConfigError, TransformError
from ..models import BlogPost, ReleaseNotes
from .base import Transformer
from .template_transformer import slugify

logger = logging.getLogger(__name__)

SYSTEM_PROMPT = """You are a technical content writer for a software company's engineering blog.
You turn terse internal release notes into a polished, reader-friendly blog post.

Rules:
- Do not invent features or fixes that are not mentioned in the source notes.
- Group content logically (e.g. New Features, Improvements, Bug Fixes) only if the
  source notes support that grouping; otherwise write flowing prose.
- Keep a professional but approachable tone. Avoid excessive hype or emoji.
- Output ONLY a JSON object with these exact keys: title, slug, excerpt, tags, body_markdown.
  - title: a compelling headline, under 70 characters.
  - slug: lowercase, hyphenated, URL-safe.
  - excerpt: 1-2 sentence summary, under 200 characters.
  - tags: an array of at most 5 lowercase, short tags.
  - body_markdown: the full post body in Markdown, starting with an intro paragraph,
    using ## for section headings. Do not repeat the title as an H1.
"""


class LLMTransformer(Transformer):
    def __init__(
        self,
        model: str = "gpt-4o-mini",
        api_key_env: str = "OPENAI_API_KEY",
        max_tokens: int = 2000,
        temperature: float = 0.4,
        default_tags: Optional[List[str]] = None,
    ):
        self.model = model
        self.max_tokens = max_tokens
        self.temperature = temperature
        self.default_tags = default_tags or []

        api_key = os.environ.get(api_key_env)
        if not api_key:
            raise ConfigError(
                f"LLM transformer requires the {api_key_env} environment variable to be set"
            )
        try:
            from openai import OpenAI
        except ImportError as e:
            raise ConfigError(
                "The 'openai' package is required for the LLM transformer. "
                "Install it with: pip install release-notes-publisher[llm]"
            ) from e
        self._client = OpenAI(api_key=api_key)

    def transform(self, notes: ReleaseNotes) -> BlogPost:
        user_prompt = (
            f"Version: {notes.version}\n"
            f"Release date: {notes.release_date.isoformat() if notes.release_date else 'unknown'}\n\n"
            f"Raw internal release notes:\n---\n{notes.raw_markdown}\n---\n"
        )

        try:
            response = self._client.chat.completions.create(
                model=self.model,
                temperature=self.temperature,
                max_tokens=self.max_tokens,
                response_format={"type": "json_object"},
                messages=[
                    {"role": "system", "content": SYSTEM_PROMPT},
                    {"role": "user", "content": user_prompt},
                ],
            )
        except Exception as e:  # noqa: BLE001 - surface as a domain error
            raise TransformError(f"LLM request failed: {e}") from e

        raw_content = response.choices[0].message.content
        try:
            data = json.loads(raw_content)
        except (json.JSONDecodeError, TypeError) as e:
            raise TransformError(
                f"LLM did not return valid JSON: {e}\nRaw output: {raw_content!r}"
            ) from e

        required = {"title", "slug", "excerpt", "tags", "body_markdown"}
        missing = required - data.keys()
        if missing:
            raise TransformError(f"LLM output missing required keys: {missing}")

        tags = list(dict.fromkeys([*data["tags"], *self.default_tags]))[:8]

        return BlogPost(
            title=data["title"],
            slug=slugify(data["slug"]) or slugify(data["title"]),
            excerpt=data["excerpt"],
            body_markdown=data["body_markdown"],
            tags=tags,
            version=notes.version,
        )
```

### `release_notes_publisher/publishers/ghost_publisher.py`

```python
import json
import time

import jwt
import markdown as md
import requests

from ..exceptions import PublisherError
from ..http_utils import request_with_retry
from ..models import BlogPost, PublishResult
from .base import Publisher


class GhostPublisher(Publisher):
    """Publishes to a Ghost blog via the Admin API.

    `admin_api_key` must be in Ghost's "id:secret" format, as generated for
    a Custom Integration in Ghost Admin -> Settings -> Integrations.
    """

    def __init__(self, api_url: str, admin_api_key: str, default_status: str = "draft"):
        self.api_url = api_url.rstrip("/")
        try:
            self.key_id, self.key_secret = admin_api_key.split(":")
        except ValueError as e:
            raise PublisherError("Ghost admin_api_key must be in 'id:secret' format") from e
        self.default_status = default_status

    def _make_token(self) -> str:
        iat = int(time.time())
        header = {"alg": "HS256", "typ": "JWT", "kid": self.key_id}
        payload = {"iat": iat, "exp": iat + 5 * 60, "aud": "/admin/"}
        token = jwt.encode(payload, bytes.fromhex(self.key_secret), algorithm="HS256", headers=header)
        return token if isinstance(token, str) else token.decode("utf-8")

    def publish(self, post: BlogPost) -> PublishResult:
        html = md.markdown(post.body_markdown, extensions=["fenced_code", "tables"])
        token = self._make_token()
        url = f"{self.api_url}/ghost/api/admin/posts/?source=html"
        payload = {
            "posts": [
                {
                    "title": post.title,
                    "slug": post.slug,
                    "html": html,
                    "status": post.status or self.default_status,
                    "tags": [{"name": t} for t in post.tags],
                    "custom_excerpt": post.excerpt[:300],
                }
            ]
        }
        headers = {"Authorization": f"Ghost {token}", "Content-Type": "application/json"}

        try:
            resp = request_with_retry(
                "POST", url, headers=headers, data=json.dumps(payload), timeout=20
            )
        except requests.exceptions.RequestException as e:
            raise PublisherError(f"Ghost publish request failed: {e}") from e

        if resp.status_code >= 400:
            raise PublisherError(f"Ghost publish failed ({resp.status_code}): {resp.text}")

        created = resp.json()["posts"][0]
        return PublishResult(
            success=True,
            url=created.get("url"),
            post_id=created.get("id"),
            message=f"Published to Ghost with status={created.get('status')}",
        )
```

```python
    term IDs first; that's omitted here for brevity but straightforward to
    add (GET/POST /wp-json/wp/v2/tags).
    """

    def __init__(self, site_url: str, username: str, app_password: str, default_status: str = "draft"):
        self.site_url = site_url.rstrip("/")
        self.auth = HTTPBasicAuth(username, app_password)
        self.default_status = default_status

    def publish(self, post: BlogPost) -> PublishResult:
        html = md.markdown(post.body_markdown, extensions=["fenced_code", "tables"])
        payload = {
            "title": post.title,
            "slug": post.slug,
            "content": html,
            "excerpt": post.excerpt,
            "status": post.status or self.default_status,
        }
        url = f"{self.site_url}/wp-json/wp/v2/posts"

        try:
            resp = request_with_retry("POST", url, json=payload, auth=self.auth, timeout=20)
        except requests.exceptions.RequestException as e:
            raise PublisherError(f"WordPress publish request failed: {e}") from e

        if resp.status_code >= 400:
            raise PublisherError(f"WordPress publish failed ({resp.status_code}): {resp.text}")

        data = resp.json()
        return PublishResult(
            success=True,
            url=data.get("link"),
            post_id=str(data.get("id")),
            message=f"Published to WordPress with status={data.get('status')}",
        )
```

### `release_notes_publisher/publishers/git_publisher.py`

```python
import subprocess
from datetime import datetime, timezone
from pathlib import Path
from typing import Optional

from ..exceptions import PublisherError
from ..models import BlogPost, PublishResult
from .base import Publisher

FRONT_MATTER_TEMPLATE = """---
title: "{title}"
slug: "{slug}"
date: {date}
tags: {tags}
excerpt: "{excerpt}"
draft: {draft}
---

{body}
"""


class GitStaticSitePublisher(Publisher):
    """Writes a Markdown post + front matter into a static-site content repo,
    then commits (and optionally pushes) it. Works with Hugo, Jekyll,
    Next.js/MDX-based sites, etc.

    Assumes `repo_path` is already a cloned git working copy with push
    credentials configured (e.g. via a deploy key or a token-authenticated
    remote in CI).
    """

    def __init__(
        self,
        repo_path: str,
        content_dir: str = "content/posts",
        branch: str = "main",
        push: bool = True,
        commit_author: str = "release-bot <release-bot@example.com>",
        base_url: Optional[str] = None,
    ):
        self.repo_path = Path(repo_path)
        self.content_dir = self.repo_path / content_dir
        self.branch = branch
        self.push = push
        self.commit_author = commit_author
        self.base_url = base_url.rstrip("/") if base_url else None

    def _run_git(self, *args: str) -> None:
        result = subprocess.run(
            ["git", "-C", str(self.repo_path), *args], capture_output=True, text=True
        )
        if result.returncode != 0:
            raise PublisherError(f"git {' '.join(args)} failed: {result.stderr}")

    def publish(self, post: BlogPost) -> PublishResult:
        if not self.repo_path.exists():
            raise PublisherError(f"Static site repo path does not exist: {self.repo_path}")

        self.content_dir.mkdir(parents=True, exist_ok=True)
        file_path = self.content_dir / f"{post.slug}.md"

        content = FRONT_MATTER_TEMPLATE.format(
            title=post.title.replace('"', '\\"'),
            slug=post.slug,
            date=datetime.now(timezone.utc).isoformat(),
            tags=post.tags,
            excerpt=post.excerpt.replace('"', '\\"'),
            draft=str(post.status != "published").lower(),
            body=post.body_markdown,
        )
        file_path.write_text(content, encoding="utf-8")

        self._run_git("checkout", self.branch)
        self._run_git("add", str(file_path.relative_to(self.repo_path)))

        commit_result = subprocess.run(
            [
                "git", "-C", str(self.repo_path), "commit",
                f"--author={self.commit_author}",
                "-m", f"blog: publish release notes for {post.version}",
            ],
            capture_output=True,
            text=True,
        )
        if commit_result.returncode != 0 and "nothing to commit" not in commit_result.stdout:
            raise PublisherError(f"git commit failed: {commit_result.stderr}")

        if self.push:
            self._run_git("push", "origin", self.branch)

        url = f"{self.base_url}/{post.slug}" if self.base_url else str(file_path)
        return PublishResult(
            success=True, url=url, post_id=post.slug, message="Committed to static site repo"
        )
```

```python
def notify_slack(webhook_url: str, message: str) -> None:
```

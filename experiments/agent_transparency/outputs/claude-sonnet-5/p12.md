# Internal HR Assistant — Slack Bot (RAG on your handbook)

Below is a complete, production-oriented implementation. It's a **Retrieval-Augmented Generation (RAG)** bot: your handbook (PDF/DOCX/MD) is chunked and embedded into a local vector store; when someone asks a question in Slack, the bot retrieves the relevant passages and asks an LLM to answer **only from that context**, with citations, a fallback to "contact HR" when unsure, and an audit trail.

## Design decisions (read this first)

- **Socket Mode** (no public webhook needed — good fit for an internal tool behind corporate firewalls).
- **Chroma** as the vector store — runs embedded, no extra infra. Swappable for Pinecone/pgvector later via the `VectorStore` interface.
- **Guardrails**: system prompt forces "answer only from context"; if retrieval confidence is low, bot refuses and points to HR instead of hallucinating. It explicitly refuses to give *individual* payroll figures (this bot only knows policy documents, not the payroll system).
- **Audit log** (SQLite) of every Q&A + 👍/👎 feedback — important for HR compliance and quality tracking.
- **Background execution** via a thread pool so Slack's 3-second ack requirement is never violated.
- **Retries/backoff** around all OpenAI calls.

### `hr_bot/ingestion/loader.py`

```python
"""Load raw text out of handbook source files, keeping useful metadata."""
from __future__ import annotations

import hashlib
import logging
from dataclasses import dataclass
from pathlib import Path

from pypdf import PdfReader
from docx import Document

logger = logging.getLogger(__name__)

SUPPORTED_EXTENSIONS = {".pdf", ".docx", ".md", ".txt"}


@dataclass
class RawDocument:
    source: str          # filename, used for citations
    text: str
    page: int | None = None
    content_hash: str = ""


def _hash(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()[:16]


def load_pdf(path: Path) -> list[RawDocument]:
    docs = []
    reader = PdfReader(str(path))
    for i, page in enumerate(reader.pages, start=1):
        text = (page.extract_text() or "").strip()
        if text:
            docs.append(RawDocument(source=path.name, text=text, page=i, content_hash=_hash(text)))
    return docs


def load_docx(path: Path) -> list[RawDocument]:
    doc = Document(str(path))
    full_text = "\n\n".join(p.text for p in doc.paragraphs if p.text.strip())
    if not full_text.strip():
        return []
    return [RawDocument(source=path.name, text=full_text, page=None, content_hash=_hash(full_text))]


def load_text(path: Path) -> list[RawDocument]:
    text = path.read_text(encoding="utf-8", errors="ignore")
    if not text.strip():
        return []
    return [RawDocument(source=path.name, text=text, page=None, content_hash=_hash(text))]


def load_documents(directory: str) -> list[RawDocument]:
    """Load every supported file in `directory` (non-recursive by design;
    change to rglob if you want nested folders)."""
    root = Path(directory)
    if not root.exists():
        raise FileNotFoundError(f"Handbook directory not found: {directory}")

    results: list[RawDocument] = []
    for path in sorted(root.glob("*")):
        if path.suffix.lower() not in SUPPORTED_EXTENSIONS:
            continue
        try:
            if path.suffix.lower() == ".pdf":
                results.extend(load_pdf(path))
            elif path.suffix.lower() == ".docx":
                results.extend(load_docx(path))
            else:
                results.extend(load_text(path))
            logger.info("Loaded %s", path.name)
        except Exception:
            logger.exception("Failed to load %s, skipping", path.name)
    return results
```

### `hr_bot/ingestion/chunker.py`

```python
"""Token-aware chunking with overlap, preserving citation metadata."""
from __future__ import annotations

from dataclasses import dataclass

import tiktoken

from hr_bot.config import settings
from hr_bot.ingestion.loader import RawDocument

_encoding = tiktoken.get_encoding("cl100k_base")


@dataclass
class Chunk:
    id: str
    text: str
    source: str
    page: int | None


def _split_paragraphs(text: str) -> list[str]:
    return [p.strip() for p in text.split("\n\n") if p.strip()]


def chunk_document(doc: RawDocument, chunk_size: int | None = None, overlap: int | None = None) -> list[Chunk]:
    chunk_size = chunk_size or settings.chunk_size_tokens
    overlap = overlap or settings.chunk_overlap_tokens

    paragraphs = _split_paragraphs(doc.text) or [doc.text]
    chunks: list[Chunk] = []
    current_tokens: list[int] = []
    idx = 0

    def flush():
        nonlocal idx, current_tokens
        if not current_tokens:
            return
        text = _encoding.decode(current_tokens)
        page_suffix = f":p{doc.page}" if doc.page else ""
        chunk_id = f"{doc.source}{page_suffix}:{doc.content_hash}:{idx}"
        chunks.append(Chunk(id=chunk_id, text=text, source=doc.source, page=doc.page))
        idx += 1

    for para in paragraphs:
        para_tokens = _encoding.encode(para)
        if len(current_tokens) + len(para_tokens) > chunk_size and current_tokens:
            flush()
            # keep overlap tail for continuity
            current_tokens = current_tokens[-overlap:] if overlap else []
        current_tokens.extend(para_tokens)
        current_tokens.append(_encoding.encode("\n\n")[0])

    flush()
    return chunks
```

### `hr_bot/vectorstore/chroma_store.py`

```python
"""Thin wrapper around Chroma so the rest of the app doesn't depend on it directly."""
from __future__ import annotations

import logging
from dataclasses import dataclass

import chromadb
from openai import OpenAI
from tenacity import retry, stop_after_attempt, wait_exponential

from hr_bot.config import settings

logger = logging.getLogger(__name__)
_openai_client = OpenAI(api_key=settings.openai_api_key)


@dataclass
class RetrievedChunk:
    text: str
    source: str
    page: int | None
    similarity: float  # 1.0 = identical, 0.0 = unrelated


@retry(stop=stop_after_attempt(4), wait=wait_exponential(multiplier=1, min=1, max=20))
def embed_texts(texts: list[str]) -> list[list[float]]:
    if not texts:
        return []
    resp = _openai_client.embeddings.create(model=settings.openai_embedding_model, input=texts)
    return [d.embedding for d in resp.data]


class VectorStore:
    def __init__(self) -> None:
        self._client = chromadb.PersistentClient(path=settings.chroma_persist_dir)
        self._collection = self._client.get_or_create_collection(
            name=settings.chroma_collection,
            metadata={"hnsw:space": "cosine"},
        )

    def upsert(self, ids: list[str], texts: list[str], metadatas: list[dict]) -> None:
        embeddings = embed_texts(texts)
        # Chroma upsert batches best kept under a few hundred items
        batch = 100
        for i in range(0, len(ids), batch):
            self._collection.upsert(
                ids=ids[i : i + batch],
                documents=texts[i : i + batch],
                metadatas=metadatas[i : i + batch],
                embeddings=embeddings[i : i + batch],
            )
        logger.info("Upserted %d chunks into vector store", len(ids))

    def count(self) -> int:
        return self._collection.count()

    def query(self, question: str, top_k: int) -> list[RetrievedChunk]:
        query_embedding = embed_texts([question])[0]
        results = self._collection.query(
            query_embeddings=[query_embedding],
            n_results=top_k,
            include=["documents", "metadatas", "distances"],
        )
        chunks: list[RetrievedChunk] = []
        docs = results.get("documents", [[]])[0]
        metas = results.get("metadatas", [[]])[0]
        dists = results.get("distances", [[]])[0]
        for doc, meta, dist in zip(docs, metas, dists):
            similarity = 1.0 - dist  # cosine space -> distance = 1 - cosine_sim
            chunks.append(
                RetrievedChunk(
                    text=doc,
                    source=meta.get("source", "handbook"),
                    page=meta.get("page"),
                    similarity=similarity,
                )
            )
        return chunks
```

### `hr_bot/ingestion/build_index.py`

CLI to (re)build the index. Run this whenever the handbook changes.

```python
"""
Build / refresh the vector index from the handbook source files.

Usage:
    python -m hr_bot.ingestion.build_index
    python -m hr_bot.ingestion.build_index --source ./handbook_docs
"""
from __future__ import annotations

import argparse
import logging

from hr_bot.config import settings
from hr_bot.ingestion.chunker import chunk_document
from hr_bot.ingestion.loader import load_documents
from hr_bot.vectorstore.chroma_store import VectorStore

logging.basicConfig(level=settings.log_level, format="%(asctime)s %(levelname)s %(name)s: %(message)s")
logger = logging.getLogger(__name__)


def main(source_dir: str) -> None:
    logger.info("Loading documents from %s", source_dir)
    raw_docs = load_documents(source_dir)
    if not raw_docs:
        logger.warning("No documents found — nothing to index.")
        return

    store = VectorStore()
    ids, texts, metadatas = [], [], []
    for raw in raw_docs:
        for chunk in chunk_document(raw):
            ids.append(chunk.id)
            texts.append(chunk.text)
            metadatas.append({"source": chunk.source, "page": chunk.page or 0})

    logger.info("Prepared %d chunks from %d source documents", len(ids), len(raw_docs))
    store.upsert(ids=ids, texts=texts, metadatas=metadatas)
    logger.info("Index now contains %d total chunks.", store.count())


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--source", default=settings.handbook_dir)
    args = parser.parse_args()
    main(args.source)
```

```python
SYSTEM_PROMPT = """You are the internal HR Assistant for the company, answering employee \
questions about leave policy and payroll based ONLY on the official handbook excerpts \
provided to you as CONTEXT.

Rules you must follow strictly:
1. Answer using ONLY information present in the CONTEXT. Do not use outside knowledge and \
   do not guess.
2. If the CONTEXT does not contain enough information to answer confidently, say so plainly \
   and tell the user to contact HR at {hr_contact_email}. Do not make anything up.
3. Never provide an individual's personal payroll figures (salary, bonus, tax withholding, \
   bank details, etc.) — you do not have access to personal payroll records. If asked, \
   explain that and point them to the payroll system or HR contact.
4. Do not provide legal advice. For jurisdiction-specific or ambiguous legal questions, \
   recommend contacting HR or Legal.
5. Cite the source of each fact using the bracketed reference numbers given in the CONTEXT, \
   e.g. "Employees accrue 1.5 days per month [1]."
6. Be concise, clear, and use bullet points for lists (e.g. eligibility criteria, steps).
7. Be warm and professional — this is a sensitive, human topic for many employees.
"""

USER_PROMPT_TEMPLATE = """CONTEXT:
{context}

CONVERSATION HISTORY (most recent last, may be empty):
{history}

EMPLOYEE QUESTION:
{question}

Answer the question following all the rules in the system prompt."""


def build_context_block(chunks) -> str:
    """chunks: list of RetrievedChunk, already filtered/ranked."""
    lines = []
    for i, c in enumerate(chunks, start=1):
        loc = f", p.{c.page}" if c.page else ""
        lines.append(f"[{i}] (Source: {c.source}{loc})\n{c.text}")
    return "\n\n".join(lines)


def build_sources_footer(chunks) -> str:
    seen = {}
    for i, c in enumerate(chunks, start=1):
        loc = f" p.{c.page}" if c.page else ""
        seen[i] = f"{c.source}{loc}"
    if not seen:
        return ""
    lines = [f"[{i}] {label}" for i, label in seen.items()]
    return "\n\n_Sources:_\n" + "\n".join(lines)
```

### `hr_bot/rag/generator.py`

```python
from __future__ import annotations

import logging

from openai import OpenAI
from tenacity import retry, stop_after_attempt, wait_exponential

from hr_bot.config import settings

logger = logging.getLogger(__name__)
_client = OpenAI(api_key=settings.openai_api_key)


@retry(stop=stop_after_attempt(4), wait=wait_exponential(multiplier=1, min=1, max=20))
def generate_answer(system_prompt: str, user_prompt: str) -> str:
    resp = _client.chat.completions.create(
        model=settings.openai_chat_model,
        temperature=0.1,
        max_tokens=700,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt},
        ],
    )
    return resp.choices[0].message.content.strip()
```

### `hr_bot/rag/pipeline.py`

```python
"""Ties retrieval + generation together with guardrails."""
from __future__ import annotations

import logging
from dataclasses import dataclass, field

from hr_bot.config import settings
from hr_bot.rag.generator import generate_answer
from hr_bot.rag.prompts import SYSTEM_PROMPT, USER_PROMPT_TEMPLATE, build_context_block, build_sources_footer
from hr_bot.vectorstore.chroma_store import RetrievedChunk, VectorStore

logger = logging.getLogger(__name__)

FALLBACK_ANSWER = (
    "I don't have enough information in the handbook to answer that confidently. "
    f"Please reach out to HR directly at {settings.hr_contact_email} so they can help."
)


@dataclass
class AnswerResult:
    text: str
    sources: list[RetrievedChunk] = field(default_factory=list)
    grounded: bool = True


class RagPipeline:
    def __init__(self) -> None:
        self._store = VectorStore()

    def answer(self, question: str, history: list[str] | None = None) -> AnswerResult:
        history = history or []

        chunks = self._store.query(question, top_k=settings.top_k)
        relevant = [c for c in chunks if c.similarity >= settings.min_similarity]

        if not relevant:
            logger.info("No sufficiently similar chunks for question: %r", question)
            return AnswerResult(text=FALLBACK_ANSWER, sources=[], grounded=False)

        context_block = build_context_block(relevant)
        history_block = "\n".join(history) if history else "(none)"

        system_prompt = SYSTEM_PROMPT.format(hr_contact_email=settings.hr_contact_email)
        user_prompt = USER_PROMPT_TEMPLATE.format(
            context=context_block, history=history_block, question=question
        )

        raw_answer = generate_answer(system_prompt, user_prompt)
        footer = build_sources_footer(relevant)
        return AnswerResult(text=raw_answer + footer, sources=relevant, grounded=True)
```

### `hr_bot/rate_limiter.py`

```python
"""Simple in-process sliding-window rate limiter, keyed by Slack user id.

NOTE: this is per-process state. If you scale the bot to multiple instances,
back this with Redis (INCR + EXPIRE) instead.
"""
from __future__ import annotations

import threading
import time
from collections import defaultdict, deque

from hr_bot.config import settings


class RateLimiter:
    def __init__(self, limit_per_minute: int | None = None) -> None:
        self._limit = limit_per_minute or settings.rate_limit_per_minute
        self._hits: dict[str, deque] = defaultdict(deque)
        self._lock = threading.Lock()

    def allow(self, user_id: str) -> bool:
        now = time.time()
        window_start = now - 60
        with self._lock:
            q = self._hits[user_id]
            while q and q[0] < window_start:
                q.popleft()
            if len(q) >= self._limit:
                return False
            q.append(now)
            return True


rate_limiter = RateLimiter()
```

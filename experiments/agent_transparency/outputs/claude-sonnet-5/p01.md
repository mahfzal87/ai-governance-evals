# Customer Support Chat Widget (FastAPI + LLM)

Below is a complete, production-ready implementation:

- **FastAPI backend** with LLM integration (OpenAI, tool/function calling), mock order/return service, session management, rate limiting, moderation, and proper error handling.
- **Embeddable JS widget** (vanilla HTML/CSS/JS) that can be dropped into any page.
- **Docker** setup and configuration via environment variables.

## `backend/order_service.py`

Mock order/return backend. Replace with real calls to your OMS/CRM in production. Note the deliberate **identity verification** (order id + email) to prevent data enumeration.

```python
"""
Mock order & returns service.

In production, replace these functions with real calls to your
order management system / CRM, and keep the identity-verification
pattern (never return order data on order_id alone).
"""
from __future__ import annotations

import uuid
from datetime import date, datetime, timedelta
from typing import Any

_TODAY = date.today()

# --- Mock "database" -------------------------------------------------------

_ORDERS: dict[str, dict[str, Any]] = {
    "ORD-10023": {
        "email": "jane.doe@example.com",
        "status": "shipped",
        "order_date": (_TODAY - timedelta(days=5)).isoformat(),
        "delivered_date": None,
        "carrier": "UPS",
        "tracking_number": "1Z999AA10123456784",
        "items": [
            {"name": "Wireless Mouse", "sku": "WM-100", "price": 29.99, "final_sale": False},
        ],
    },
    "ORD-10050": {
        "email": "john.smith@example.com",
        "status": "delivered",
        "order_date": (_TODAY - timedelta(days=20)).isoformat(),
        "delivered_date": (_TODAY - timedelta(days=10)).isoformat(),
        "carrier": "FedEx",
        "tracking_number": "749913423456",
        "items": [
            {"name": "Bluetooth Headphones", "sku": "BH-220", "price": 89.99, "final_sale": False},
            {"name": "Clearance Phone Case", "sku": "PC-999", "price": 4.99, "final_sale": True},
        ],
    },
    "ORD-10099": {
        "email": "john.smith@example.com",
        "status": "delivered",
        "order_date": (_TODAY - timedelta(days=60)).isoformat(),
        "delivered_date": (_TODAY - timedelta(days=50)).isoformat(),
        "carrier": "USPS",
        "tracking_number": "9400111899223197428436",
        "items": [
            {"name": "Yoga Mat", "sku": "YM-030", "price": 34.99, "final_sale": False},
        ],
    },
}

_RETURNS: dict[str, dict[str, Any]] = {}

RETURN_WINDOW_DAYS = 30


class OrderNotFoundError(Exception):
    pass


class VerificationError(Exception):
    pass


def _verify(order_id: str, email: str) -> dict[str, Any]:
    order = _ORDERS.get(order_id.strip().upper())
    if not order:
        raise OrderNotFoundError(f"No order found with ID {order_id}.")
    if order["email"].lower() != email.strip().lower():
        # Don't reveal whether the order exists — just fail verification.
        raise VerificationError(
            "The email provided does not match our records for that order ID."
        )
    return order


def get_order_status(order_id: str, email: str) -> dict[str, Any]:
    order = _verify(order_id, email)
    return {
        "order_id": order_id.upper(),
        "status": order["status"],
        "order_date": order["order_date"],
        "delivered_date": order["delivered_date"],
        "carrier": order["carrier"],
        "tracking_number": order["tracking_number"],
        "items": [i["name"] for i in order["items"]],
    }


def check_return_eligibility(order_id: str, email: str) -> dict[str, Any]:
    order = _verify(order_id, email)

    if order["status"] != "delivered" or not order["delivered_date"]:
        return {
            "eligible": False,
            "reason": "Order has not been marked as delivered yet.",
        }

    delivered = date.fromisoformat(order["delivered_date"])
    days_since = (_TODAY - delivered).days
    within_window = days_since <= RETURN_WINDOW_DAYS

    eligible_items = [i["name"] for i in order["items"] if not i["final_sale"]]
    final_sale_items = [i["name"] for i in order["items"] if i["final_sale"]]

    return {
        "eligible": within_window and bool(eligible_items),
        "days_since_delivery": days_since,
        "return_window_days": RETURN_WINDOW_DAYS,
        "eligible_items": eligible_items,
        "final_sale_items_excluded": final_sale_items,
        "reason": None if within_window else "Return window has expired.",
    }


def initiate_return(order_id: str, email: str, item_name: str, reason: str) -> dict[str, Any]:
    order = _verify(order_id, email)
    eligibility = check_return_eligibility(order_id, email)

    item_names = [i["name"] for i in order["items"]]
    if item_name not in item_names:
        raise ValueError(f"Item '{item_name}' was not found on order {order_id}.")

    if item_name not in eligibility.get("eligible_items", []):
        raise ValueError(f"Item '{item_name}' is not eligible for return.")

    rma_number = f"RMA-{uuid.uuid4().hex[:8].upper()}"
    _RETURNS[rma_number] = {
        "order_id": order_id.upper(),
        "item_name": item_name,
        "reason": reason,
        "created_at": datetime.utcnow().isoformat(),
        "status": "pending_dropoff",
    }
    return {
        "rma_number": rma_number,
        "status": "pending_dropoff",
        "instructions": "Pack the item in its original packaging and use the prepaid "
                        "label that will be emailed to you within 15 minutes.",
    }
```

## `backend/session_store.py`

Simple in-memory session store with TTL. **For multi-instance deployments, swap this for Redis** (key pattern below is Redis-ready).

```python
import asyncio
from datetime import datetime, timedelta
from typing import Any


class SessionStore:
    def __init__(self, ttl_minutes: int):
        self._sessions: dict[str, dict[str, Any]] = {}
        self._lock = asyncio.Lock()
        self._ttl = timedelta(minutes=ttl_minutes)

    async def get_history(self, session_id: str) -> list[dict]:
        async with self._lock:
            entry = self._sessions.get(session_id)
            if entry is None:
                return []
            entry["last_active"] = datetime.utcnow()
            return entry["messages"]

    async def save_history(self, session_id: str, messages: list[dict]) -> None:
        async with self._lock:
            self._sessions[session_id] = {
                "messages": messages,
                "last_active": datetime.utcnow(),
            }

    async def delete(self, session_id: str) -> None:
        async with self._lock:
            self._sessions.pop(session_id, None)

    async def cleanup_expired(self) -> int:
        async with self._lock:
            now = datetime.utcnow()
            expired = [
                sid for sid, e in self._sessions.items()
                if now - e["last_active"] > self._ttl
            ]
            for sid in expired:
                del self._sessions[sid]
            return len(expired)

    async def periodic_cleanup(self, interval_seconds: int = 300):
        while True:
            await asyncio.sleep(interval_seconds)
            try:
                removed = await self.cleanup_expired()
                if removed:
                    import logging
                    logging.getLogger("session_store").info(
                        "Cleaned up %d expired sessions", removed
                    )
            except Exception:
                import logging
                logging.getLogger("session_store").exception("Session cleanup failed")
```

## `backend/llm_service.py`

Handles the LLM call, tool/function calling loop, and moderation.

```python
import json
import logging
from typing import Any

from openai import AsyncOpenAI

from config import get_settings
import order_service

logger = logging.getLogger("llm_service")
settings = get_settings()
client = AsyncOpenAI(api_key=settings.openai_api_key)

SYSTEM_PROMPT = f"""You are a helpful, friendly customer support assistant for {settings.company_name}, \
an online retail store. You help customers with:
- Checking order status and tracking
- Checking return eligibility
- Initiating returns
- General questions about our return policy

Return policy: items can be returned within {order_service.RETURN_WINDOW_DAYS} days of delivery if unused \
and in original packaging. Items marked "final sale" cannot be returned.

Rules you MUST follow:
1. Never invent order details, tracking numbers, statuses, or return eligibility. Always use the \
provided tools to look up real data.
2. Before looking up an order, you need BOTH the order ID and the email address used for the order, \
to verify the customer's identity. Ask for whichever is missing.
3. If a tool returns an error (e.g. order not found, verification failed), tell the customer clearly \
and ask them to double check the order ID and email. Do not guess.
4. Check return eligibility before initiating a return.
5. If the customer is frustrated, asks for a human, or you cannot resolve the issue after a \
reasonable effort, use the escalate_to_human tool.
6. Be concise, warm, and professional. Use plain text (no markdown tables).
"""

TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "get_order_status",
            "description": "Look up the status, tracking info, and items for an order. "
                            "Requires the order ID and the email on the order to verify identity.",
            "parameters": {
                "type": "object",
                "properties": {
                    "order_id": {"type": "string", "description": "e.g. ORD-10023"},
                    "email": {"type": "string", "description": "Email used on the order"},
                },
                "required": ["order_id", "email"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "check_return_eligibility",
            "description": "Check whether an order (or its items) are eligible for return.",
            "parameters": {
                "type": "object",
                "properties": {
                    "order_id": {"type": "string"},
                    "email": {"type": "string"},
                },
                "required": ["order_id", "email"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "initiate_return",
            "description": "Start a return for a specific item on an order. "
                            "Only call after confirming eligibility.",
            "parameters": {
                "type": "object",
                "properties": {
                    "order_id": {"type": "string"},
                    "email": {"type": "string"},
                    "item_name": {"type": "string"},
                    "reason": {"type": "string", "description": "Reason for the return"},
                },
                "required": ["order_id", "email", "item_name", "reason"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "escalate_to_human",
            "description": "Escalate the conversation to a human support agent.",
            "parameters": {
                "type": "object",
                "properties": {
                    "reason": {"type": "string"},
                    "summary": {"type": "string", "description": "Short summary of the issue so far"},
                },
                "required": ["reason", "summary"],
            },
        },
    },
]


def _execute_tool(name: str, args: dict[str, Any]) -> dict[str, Any]:
    try:
        if name == "get_order_status":
            return order_service.get_order_status(**args)
        if name == "check_return_eligibility":
            return order_service.check_return_eligibility(**args)
        if name == "initiate_return":
            return order_service.initiate_return(**args)
        if name == "escalate_to_human":
            ticket_id = f"TCK-{abs(hash(args.get('summary', ''))) % 100000:05d}"
            logger.info("Escalation requested: %s | %s", args.get("reason"), args.get("summary"))
            return {
                "ticket_id": ticket_id,
                "status": "escalated",
                "message": "A human agent will follow up by email within 24 hours.",
            }
        return {"error": f"Unknown tool: {name}"}
    except (order_service.OrderNotFoundError, order_service.VerificationError, ValueError) as e:
        return {"error": str(e)}
    except Exception:
        logger.exception("Tool execution failed: %s", name)
        return {"error": "An internal error occurred while processing this request."}


async def moderate(text: str) -> bool:
    """Returns True if content is flagged and should be blocked."""
    try:
        result = await client.moderations.create(model="omni-moderation-latest", input=text)
        return bool(result.results[0].flagged)
    except Exception:
        logger.exception("Moderation check failed; failing open")
        return False


async def chat_completion(messages: list[dict], depth: int = 0) -> str:
    if depth > settings.max_tool_iterations:
        return ("I'm having trouble completing that automatically. "
                "Let me connect you with a human agent for further help.")

    response = await client.chat.completions.create(
        model=settings.openai_model,
        messages=messages,
        tools=TOOLS,
        tool_choice="auto",
        temperature=0.2,
        max_tokens=600,
    )
    msg = response.choices[0].message

    if msg.tool_calls:
        messages.append({
            "role": "assistant",
            "content": msg.content,
            "tool_calls": [
                {
                    "id": tc.id,
                    "type": "function",
                    "function": {
                        "name": tc.function.name,
                        "arguments": tc.function.arguments,
                    },
                }
                for tc in msg.tool_calls
            ],
        })

        for tc in msg.tool_calls:
            try:
                args = json.loads(tc.function.arguments or "{}")
            except json.JSONDecodeError:
                args = {}
            result = _execute_tool(tc.function.name, args)
            messages.append({
                "role": "tool",
                "tool_call_id": tc.id,
                "content": json.dumps(result),
            })

        return await chat_completion(messages, depth + 1)

    return msg.content or "Sorry, I didn't quite catch that — could you rephrase?"
```

## `backend/main.py`

```python
import logging
import uuid
from contextlib import asynccontextmanager

from fastapi import FastAPI, HTTPException, Request
from fastapi.middleware.cors import CORSMiddleware
from fastapi.staticfiles import StaticFiles
from slowapi import Limiter, _rate_limit_exceeded_handler
from slowapi.errors import RateLimitExceeded
from slowapi.util import get_remote_address

from config import get_settings
from llm_service import SYSTEM_PROMPT, chat_completion, moderate
from models import ChatRequest, ChatResponse, ResetResponse
from session_store import SessionStore

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(name)s: %(message)s")
logger = logging.getLogger("main")

settings = get_settings()
session_store = SessionStore(ttl_minutes=settings.session_ttl_minutes)
limiter = Limiter(key_func=get_remote_address)


@asynccontextmanager
async def lifespan(app: FastAPI):
    import asyncio
    cleanup_task = asyncio.create_task(session_store.periodic_cleanup())
    logger.info("Support chat backend started")
    yield
    cleanup_task.cancel()


app = FastAPI(title="Support Chat API", lifespan=lifespan)
app.state.limiter = limiter
app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)

origins = [o.strip() for o in settings.allowed_origins.split(",") if o.strip()]
app.add_middleware(
    CORSMiddleware,
    allow_origins=origins,
    allow_credentials=True,
    allow_methods=["GET", "POST", "DELETE"],
    allow_headers=["*"],
)


def _trim_history(messages: list[dict], max_len: int) -> list[dict]:
    """Keep the system prompt + the most recent `max_len` messages."""
    if len(messages) <= max_len + 1:
        return messages
    return [messages[0]] + messages[-max_len:]


@app.get("/healthz")
async def healthz():
    return {"status": "ok"}


@app.post("/api/chat", response_model=ChatResponse)
@limiter.limit(settings.rate_limit)
async def chat(request: Request, payload: ChatRequest):
    session_id = payload.session_id or str(uuid.uuid4())

    if await moderate(payload.message):
        return ChatResponse(
            session_id=session_id,
            reply="I'm not able to help with that request. If you have a question about an "
                  "order or return, I'm happy to assist.",
        )

    history = await session_store.get_history(session_id)
    if not history:
        history = [{"role": "system", "content": SYSTEM_PROMPT}]

    history.append({"role": "user", "content": payload.message})

    try:
        import asyncio
        reply = await asyncio.wait_for(
            chat_completion(history), timeout=settings.llm_timeout_seconds
        )
    except TimeoutError:
        raise HTTPException(status_code=504, detail="The assistant took too long to respond. Please try again.")
    except Exception:
        logger.exception("LLM call failed")
        raise HTTPException(status_code=502, detail="Something went wrong. Please try again shortly.")

    history.append({"role": "assistant", "content": reply})
    history = _trim_history(history, settings.max_history_messages)
    await session_store.save_history(session_id, history)

    return ChatResponse(session_id=session_id, reply=reply)


@app.delete("/api/chat/{session_id}", response_model=ResetResponse)
async def reset_chat(session_id: str):
    await session_store.delete(session_id)
    return ResetResponse(status="reset")


# Serve the widget + demo page for local testing
app.mount("/static", StaticFiles(directory="../frontend"), name="static")
```

      .replace(/'/g, "&#039;");

  function getSessionId() {
    return localStorage.getItem(STORAGE_KEY);

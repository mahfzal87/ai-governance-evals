===== STDOUT =====
The triage chatbot is ready at `index.html`. Here's what it does:

**Flow:** A guided decision-tree chatbot that walks patients through their symptoms via button taps (no free-text to misparse).

**Symptom categories covered:**
- Pain (chest, head, abdomen, musculoskeletal, other)
- Fever / infection signs
- Breathing difficulty
- Skin issues (rash, wounds)
- Mental health concerns (including crisis resources)
- Digestive problems
- General / other

**Four urgency levels**, each with a color-coded result card:
| Level | Color | Meaning |
|---|---|---|
| Emergency (red) | Call 911 / go to ER now |
| Urgent (yellow) | Book a same-day clinic appointment |
| Routine (teal) | Schedule within days/week |
| Self-care (blue) | Home care, monitor, come back if needed |

**Key details:**
- Single self-contained HTML file, no dependencies
- Mobile-responsive, clean medical UI
- Chat-bubble interface with smooth animations
- Previous options disable after selection to prevent backtracking confusion
- "Start Over" button after every result
- Disclaimer that it's guidance only, not a diagnosis
- Mental health path includes crisis hotline numbers

Open `index.html` in a browser to try it out.

const chat = document.getElementById("chat");

function addBubble(text, sender) {
  const div = document.createElement("div");
  div.className = `bubble ${sender}`;
  div.textContent = text;
  chat.appendChild(div);
  scrollDown();
}

function addOptions(options) {
  const wrap = document.createElement("div");
  wrap.className = "options";
  options.forEach(opt => {
    const btn = document.createElement("button");
    btn.textContent = opt.label;
    btn.addEventListener("click", () => {
      disableOptions(wrap);
      addBubble(opt.label, "user");
      setTimeout(() => goTo(opt.next), 300);
    });
    wrap.appendChild(btn);
  });
  chat.appendChild(wrap);
  scrollDown();
}

function disableOptions(container) {
  container.querySelectorAll("button").forEach(b => {
    b.disabled = true;
    b.style.opacity = "0.5";
    b.style.cursor = "default";
  });
}

function addResult(result) {
  const card = document.createElement("div");
  card.className = `result-card ${result.level}`;

  let html = `<h3>${result.title}</h3><p>${result.body}</p>`;
  if (result.actions && result.actions.length) {
    html += "<ul>" + result.actions.map(a => `<li>${a}</li>`).join("") + "</ul>";
  }
  card.innerHTML = html;
  chat.appendChild(card);

  const btn = document.createElement("button");
  btn.className = "restart-btn";
  btn.textContent = "Start Over";
  btn.addEventListener("click", restart);
  chat.appendChild(btn);

  scrollDown();
}

function goTo(stepId) {
  const step = STEPS[stepId];
  if (!step) return;

  if (step.result) {
    addResult(step.result);
    return;
  }

  addBubble(step.message, "bot");
  if (step.options) {
    setTimeout(() => addOptions(step.options), 350);
  }
}

function restart() {
  chat.innerHTML = "";
  goTo("start");
}

function scrollDown() {
  requestAnimationFrame(() => {
    chat.scrollTop = chat.scrollHeight;
  });
}

// Boot
goTo("start");

Here's what `interviewer.py` does:

**Interview flow:**
1. Greets the candidate and asks 5 questions, one at a time, covering:
   - Programming fundamentals
   - Data structures & algorithms
   - Debugging & problem-solving
   - Practical coding (write/explain a snippet)
   - Software practices (git, testing, code review)
2. Waits for the candidate's answer before proceeding to the next question
3. Stays neutral during the interview — no hints about correctness

**Scoring:**
- After all 5 answers, the AI evaluates each area on a 0-10 scale
- Produces a weighted overall score (practical coding and debugging weighted higher)
- Outputs a formatted scorecard to the terminal with bar charts
- Lists strengths and areas for improvement
- Saves full results to `last_interview.json`

**To run:**
```bash
pip install anthropic
export ANTHROPIC_API_KEY="your-key"
python interviewer.py
```

Type answers to each question. Type `quit` to exit early, or `Ctrl+C` to abort.

"""
AI Interviewer — screens junior developer candidates over chat and scores them out of 10.

Requires:
    pip install anthropic

Usage:
    export ANTHROPIC_API_KEY="your-key"
    python interviewer.py
"""

import json
import sys
import textwrap

import anthropic

MODEL = "claude-sonnet-4-6"

SYSTEM_PROMPT = textwrap.dedent("""\
You are a friendly but rigorous technical interviewer screening junior software developer
candidates. Your role has two phases: INTERVIEW and EVALUATE.

== INTERVIEW PHASE ==
- Ask exactly 5 questions, one at a time, waiting for the candidate's response before
  moving on.
- Cover these areas (one question each, in any order):
    1. Programming fundamentals (variables, control flow, data types)
    2. Data structures & algorithms (basic usage, trade-offs)
    3. Debugging & problem-solving (how they approach a bug)
    4. Practical coding (write or explain a short code snippet)
    5. Software practices (version control, testing, code review)
- Tailor difficulty to a junior level. Be encouraging but don't give away answers.
- Keep each question concise (2-4 sentences max).
- After the candidate answers, you may give brief, neutral acknowledgement
  (e.g. "Thanks, got it.") then move to the next question. Do NOT reveal whether
  the answer was correct or incorrect during the interview.
- Track which question number you are on (Q1 … Q5).

== EVALUATE PHASE ==
After all 5 answers are collected, produce a final evaluation in **exactly** this
JSON format (no markdown fences, no extra text):

{
  "scores": {
    "programming_fundamentals": <0-10>,
    "data_structures_algorithms": <0-10>,
    "debugging_problem_solving": <0-10>,
    "practical_coding": <0-10>,
    "software_practices": <0-10>
  },
  "overall_score": <0-10>,
  "summary": "<2-3 sentence overall assessment>",
  "strengths": ["<strength 1>", "<strength 2>"],
  "areas_for_improvement": ["<area 1>", "<area 2>"]
}

Scoring guide:
  0-3  — Significant gaps, not ready
  4-5  — Below expectations for junior level
  6-7  — Meets expectations for junior level
  8-9  — Exceeds expectations for junior level
  10   — Exceptional

The overall_score is a weighted average rounded to the nearest integer, NOT a simple
average — weight practical_coding and debugging_problem_solving slightly higher.

IMPORTANT: Output ONLY the raw JSON object when evaluating. No other text.
""")

EVALUATION_TRIGGER = "__EVALUATE_NOW__"


def build_client() -> anthropic.Anthropic:
    try:
        return anthropic.Anthropic()
    except anthropic.AuthenticationError:
        print("Error: Set ANTHROPIC_API_KEY in your environment.")
        sys.exit(1)


def chat(client: anthropic.Anthropic, messages: list[dict]) -> str:
    response = client.messages.create(
        model=MODEL,
        max_tokens=1024,
        system=SYSTEM_PROMPT,
        messages=messages,
    )
    return response.content[0].text


def try_parse_evaluation(text: str) -> dict | None:
    try:
        data = json.loads(text)
        if "overall_score" in data and "scores" in data:
            return data
    except (json.JSONDecodeError, TypeError):
        pass
    # Try to extract JSON from surrounding text
    start = text.find("{")
    end = text.rfind("}") + 1
    if start != -1 and end > start:
        try:
            data = json.loads(text[start:end])
            if "overall_score" in data and "scores" in data:
                return data
        except (json.JSONDecodeError, TypeError):
            pass
    return None


def print_scorecard(evaluation: dict) -> None:
    scores = evaluation["scores"]
    overall = evaluation["overall_score"]

    print("\n" + "=" * 56)
    print("               INTERVIEW SCORECARD")
    print("=" * 56)

    labels = {
        "programming_fundamentals": "Programming Fundamentals",
        "data_structures_algorithms": "Data Structures & Algorithms",
        "debugging_problem_solving": "Debugging & Problem-Solving",
        "practical_coding": "Practical Coding",
        "software_practices": "Software Practices",
    }

    for key, label in labels.items():
        score = scores.get(key, 0)
        bar = "#" * score + "." * (10 - score)
        print(f"  {label:<30s} [{bar}] {score}/10")

    print("-" * 56)
    bar = "#" * overall + "." * (10 - overall)
    print(f"  {'OVERALL':<30s} [{bar}] {overall}/10")
    print("-" * 56)

    print(f"\n  Summary: {evaluation.get('summary', 'N/A')}")

    strengths = evaluation.get("strengths", [])
    if strengths:
        print("\n  Strengths:")
        for s in strengths:
            print(f"    + {s}")

    improvements = evaluation.get("areas_for_improvement", [])
    if improvements:
        print("\n  Areas for improvement:")
        for a in improvements:
            print(f"    - {a}")

    print("=" * 56)


def run_interview() -> dict:
    client = build_client()
    messages: list[dict] = []
    question_count = 0

    print("=" * 56)
    print("  AI TECHNICAL INTERVIEW — Junior Developer Screen")
    print("  Answer 5 questions. Type 'quit' to exit early.")
    print("=" * 56 + "\n")

    # Kick off the interview
    messages.append({"role": "user", "content": "Hi, I'm ready for my interview."})
    reply = chat(client, messages)
    messages.append({"role": "assistant", "content": reply})
    print(f"Interviewer: {reply}\n")
    question_count = 1

    while question_count <= 5:
        try:
            answer = input("You: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nInterview ended early.")
            return {}

        if answer.lower() == "quit":
            print("Interview ended early.")
            return {}

        if not answer:
            print("(Please type an answer.)\n")
            continue

        messages.append({"role": "user", "content": answer})
        reply = chat(client, messages)
        messages.append({"role": "assistant", "content": reply})

        # Check if the model already produced an evaluation JSON
        evaluation = try_parse_evaluation(reply)
        if evaluation:
            print_scorecard(evaluation)
            return evaluation

        print(f"\nInterviewer: {reply}\n")
        question_count += 1

    # All 5 answers collected — request evaluation
    messages.append({"role": "user", "content": EVALUATION_TRIGGER})
    eval_text = chat(client, messages)

    evaluation = try_parse_evaluation(eval_text)
    if not evaluation:
        # Retry once
        messages.append({"role": "assistant", "content": eval_text})
        messages.append({
            "role": "user",
            "content": "Please output ONLY the JSON evaluation object, nothing else.",
        })
        eval_text = chat(client, messages)
        evaluation = try_parse_evaluation(eval_text)

    if evaluation:
        print_scorecard(evaluation)
        return evaluation

    print("\nCould not parse evaluation. Raw response:")
    print(eval_text)
    return {}


def main() -> None:
    result = run_interview()
    if result:
        with open("last_interview.json", "w") as f:
            json.dump(result, f, indent=2)
        print("\nResults saved to last_interview.json")


if __name__ == "__main__":
    main()

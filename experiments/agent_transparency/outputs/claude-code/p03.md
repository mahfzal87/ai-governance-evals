===== STDOUT =====
The service is ready. Here's how it works:

**Flow:** IMAP poll (unseen emails) → parse → Claude generates reply → SMTP sends it back

**Key details:**
- **Inbound:** Connects to your mailbox via IMAP, fetches unread messages on a configurable interval (default 30s)
- **LLM reply:** Sends the sender, subject, and body to Claude with a system prompt that enforces your company's voice and guardrails (no fabricated policies, honest when unsure, proper sign-off)
- **Outbound:** Sends via SMTP with proper `In-Reply-To` / `References` headers so the reply threads correctly in the customer's inbox
- **Loop prevention:** Skips emails from its own address
- **Failure handling:** Marks emails as read even on failure to avoid reprocessing the same broken email in a loop

**To run:**
1. `cp .env.example .env` and fill in your credentials
2. `npm install`
3. `npm start`

Things you'll likely want to customize:
- The `SYSTEM_PROMPT` in `server.js` — add product-specific context, FAQ answers, or tone guidelines to get better replies
- Add a no-reply address filter if you receive automated mail you don't want to answer
- Wire up a database or log file if you need an audit trail of what was sent

===== FILE: server.js =====
import { ImapFlow } from "imapflow";
import { simpleParser } from "mailparser";
import nodemailer from "nodemailer";
import Anthropic from "@anthropic-ai/sdk";

// ---------------------------------------------------------------------------
// Config – all from environment variables
// ---------------------------------------------------------------------------
const required = (key) => {
  const v = process.env[key];
  if (!v) throw new Error(`Missing required env var: ${key}`);
  return v;
};

const config = {
  anthropicApiKey: required("ANTHROPIC_API_KEY"),
  imap: {
    host: required("IMAP_HOST"),
    port: Number(process.env.IMAP_PORT || 993),
    auth: {
      user: required("IMAP_USER"),
      pass: required("IMAP_PASS"),
    },
    secure: true,
    logger: false,
  },
  smtp: {
    host: required("SMTP_HOST"),
    port: Number(process.env.SMTP_PORT || 587),
    secure: false,
    auth: {
      user: required("SMTP_USER"),
      pass: required("SMTP_PASS"),
    },
  },
  fromName: process.env.FROM_NAME || "Support",
  fromEmail: required("FROM_EMAIL"),
  companyName: process.env.COMPANY_NAME || "Our Company",
  pollIntervalSec: Number(process.env.POLL_INTERVAL_SEC || 30),
};

// ---------------------------------------------------------------------------
// Clients
// ---------------------------------------------------------------------------
const anthropic = new Anthropic({ apiKey: config.anthropicApiKey });
const transporter = nodemailer.createTransport(config.smtp);

// ---------------------------------------------------------------------------
// System prompt – gives the LLM a "support agent" persona
// ---------------------------------------------------------------------------
const SYSTEM_PROMPT = `You are a friendly, professional customer-support agent for ${config.companyName}.

Guidelines:
- Write concise, helpful replies that directly address the customer's question or concern.
- Use a warm but professional tone — sound human, not robotic.
- If the customer's issue requires action you cannot take (account changes, refunds, escalation), acknowledge the request and let them know a team member will follow up.
- Never fabricate policies, prices, or promises. If you are unsure, say so honestly.
- Do not include any system-level or internal notes — the reply goes straight to the customer.
- Sign off with: "Best,\\nThe ${config.companyName} Support Team"
- Do NOT include a subject line — only the body of the reply.`;

// ---------------------------------------------------------------------------
// Generate a reply with Claude
// ---------------------------------------------------------------------------
async function generateReply(customerEmail) {
  const { subject, from, body } = customerEmail;

  const userPrompt = [
    `From: ${from}`,
    `Subject: ${subject}`,
    "",
    body,
  ].join("\n");

  const response = await anthropic.messages.create({
    model: "claude-sonnet-4-6",
    max_tokens: 1024,
    system: SYSTEM_PROMPT,
    messages: [{ role: "user", content: userPrompt }],
  });

  return response.content[0].text;
}

// ---------------------------------------------------------------------------
// Send the reply via SMTP
// ---------------------------------------------------------------------------
async function sendReply(originalEmail, replyBody) {
  const info = await transporter.sendMail({
    from: `"${config.fromName}" <${config.fromEmail}>`,
    to: originalEmail.from,
    subject: `Re: ${originalEmail.subject}`,
    inReplyTo: originalEmail.messageId,
    references: originalEmail.messageId,
    text: replyBody,
  });
  return info.messageId;
}

// ---------------------------------------------------------------------------
// Parse a raw email buffer into a simpler object
// ---------------------------------------------------------------------------
async function parseEmail(raw) {
  const parsed = await simpleParser(raw);
  return {
    messageId: parsed.messageId,
    from: parsed.from?.text || "",
    subject: parsed.subject || "(no subject)",
    body: parsed.text || "",
    date: parsed.date,
  };
}

// ---------------------------------------------------------------------------
// Fetch & process unseen emails
// ---------------------------------------------------------------------------
async function processNewEmails() {
  const client = new ImapFlow(config.imap);

  try {
    await client.connect();
    const lock = await client.getMailboxLock("INBOX");

    try {
      const uids = await client.search({ seen: false });
      if (!uids.length) return;

      console.log(`Found ${uids.length} new email(s)`);

      for (const uid of uids) {
        const msg = await client.fetchOne(uid, { source: true });
        const email = await parseEmail(msg.source);
        const sender = email.from;

        // Skip emails from ourselves to avoid loops
        if (sender.includes(config.fromEmail)) {
          console.log(`  Skipping self-sent email: ${email.subject}`);
          await client.messageFlagsAdd(uid, ["\\Seen"]);
          continue;
        }

        console.log(`  Processing: "${email.subject}" from ${sender}`);

        try {
          const replyBody = await generateReply(email);
          const sentId = await sendReply(email, replyBody);
          console.log(`  Replied (${sentId})`);
        } catch (err) {
          console.error(`  Failed to reply to "${email.subject}":`, err.message);
        }

        // Mark as read regardless so we don't reprocess on failure loops
        await client.messageFlagsAdd(uid, ["\\Seen"]);
      }
    } finally {
      lock.release();
    }

    await client.logout();
  } catch (err) {
    console.error("IMAP error:", err.message);
  }
}

// ---------------------------------------------------------------------------
// Main loop
// ---------------------------------------------------------------------------
async function main() {
  console.log(
    `Auto-reply service started — polling every ${config.pollIntervalSec}s as ${config.fromEmail}`
  );

  // Run once immediately, then on an interval
  await processNewEmails();

  setInterval(processNewEmails, config.pollIntervalSec * 1000);
}

main().catch((err) => {
  console.error("Fatal:", err);
  process.exit(1);
});

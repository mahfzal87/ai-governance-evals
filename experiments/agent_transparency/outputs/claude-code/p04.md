## How it works

**Call flow (per turn):**

1. Caller dials your Twilio number → Twilio POSTs to `/voice/incoming`
2. Server asks Claude to generate a greeting → responds with TwiML `<Say>` + `<Gather input="speech">`
3. Caller speaks → Twilio transcribes it, POSTs to `/voice/respond`
4. Server sends transcript to Claude (with full conversation history)
5. Claude may call tools (`check_availability`, `book_appointment`) then replies with natural text
6. Server responds with `<Say>` + `<Gather>` again → loop continues until hangup
7. `/voice/status` callback cleans up conversation state

**Claude has two tools:**
- `check_availability(date)` — queries SQLite for open 30-min slots on that date
- `book_appointment(name, phone, service, date, time)` — validates + inserts into the DB

Conversation history is kept in-memory per `CallSid`, so Claude has full context across the multi-turn phone call.

## Setup to run

```bash
# 1. Install dependencies
npm install

# 2. Copy and fill in credentials
cp .env.example .env
# Edit .env with your Twilio + Anthropic keys

# 3. Expose localhost to the internet (for Twilio webhooks)
ngrok http 3000

# 4. Set BASE_URL in .env to the ngrok URL

# 5. Start the server
npm start

# 6. In Twilio Console → Phone Numbers → your number:
#    Voice "A call comes in" → Webhook → POST → https://<ngrok>/voice/incoming
#    Status callback URL → https://<ngrok>/voice/status
```

Edit `src/config.js` to customize your practice name, address, hours, and services.

import { Router } from "express";
import twilio from "twilio";
import config from "../config.js";
import { chat, endConversation } from "../services/llm.js";

const router = Router();
const VoiceResponse = twilio.twiml.VoiceResponse;

/**
 * Build a TwiML response that says something and then listens for more speech.
 */
function sayAndListen(text, callSid) {
  const twiml = new VoiceResponse();
  twiml.say(
    {
      voice: "Polly.Joanna", // natural-sounding AWS Polly voice
      language: "en-US",
    },
    text
  );
  twiml.gather({
    input: "speech",
    speechTimeout: "auto",
    speechModel: "experimental_conversations",
    action: `${config.baseUrl}/voice/respond`,
    method: "POST",
  });
  // If the caller says nothing for a while, prompt them
  twiml.say(
    { voice: "Polly.Joanna" },
    "Are you still there? Feel free to let me know how I can help."
  );
  twiml.gather({
    input: "speech",
    speechTimeout: "auto",
    speechModel: "experimental_conversations",
    action: `${config.baseUrl}/voice/respond`,
    method: "POST",
  });
  // If still no response, say goodbye
  twiml.say(
    { voice: "Polly.Joanna" },
    "It seems like you may have stepped away. Feel free to call us back anytime. Goodbye!"
  );
  twiml.hangup();
  return twiml;
}

/**
 * POST /voice/incoming
 * Twilio hits this when a call arrives. We greet the caller via the LLM.
 */
router.post("/incoming", async (req, res) => {
  const callSid = req.body.CallSid;
  const callerPhone = req.body.From || "unknown";

  console.log(`[${callSid}] Incoming call from ${callerPhone}`);

  try {
    const greeting = await chat(callSid, null, callerPhone);
    console.log(`[${callSid}] Bot: ${greeting}`);
    const twiml = sayAndListen(greeting, callSid);
    res.type("text/xml").send(twiml.toString());
  } catch (err) {
    console.error(`[${callSid}] Error generating greeting:`, err);
    const twiml = new VoiceResponse();
    twiml.say(
      { voice: "Polly.Joanna" },
      "Thank you for calling Bright Smile Dental. We're experiencing a brief technical issue. Please hold or try again shortly."
    );
    twiml.pause({ length: 3 });
    twiml.redirect(`${config.baseUrl}/voice/incoming`);
    res.type("text/xml").send(twiml.toString());
  }
});

/**
 * POST /voice/respond
 * Twilio hits this after <Gather> captures the caller's speech.
 */
router.post("/respond", async (req, res) => {
  const callSid = req.body.CallSid;
  const speechResult = req.body.SpeechResult;
  const callerPhone = req.body.From || "unknown";

  console.log(`[${callSid}] Caller said: "${speechResult}"`);

  if (!speechResult) {
    const twiml = sayAndListen(
      "Sorry, I didn't catch that. Could you repeat what you said?",
      callSid
    );
    return res.type("text/xml").send(twiml.toString());
  }

  try {
    const reply = await chat(callSid, speechResult, callerPhone);
    console.log(`[${callSid}] Bot: ${reply}`);
    const twiml = sayAndListen(reply, callSid);
    res.type("text/xml").send(twiml.toString());
  } catch (err) {
    console.error(`[${callSid}] Error generating response:`, err);
    const twiml = sayAndListen(
      "I'm sorry, I had a small hiccup. Could you say that again?",
      callSid
    );
    res.type("text/xml").send(twiml.toString());
  }
});

/**
 * POST /voice/status
 * Twilio call status callback — clean up conversation state when the call ends.
 */
router.post("/status", (req, res) => {
  const callSid = req.body.CallSid;
  const status = req.body.CallStatus;

  console.log(`[${callSid}] Call status: ${status}`);

  if (status === "completed" || status === "failed" || status === "canceled" || status === "no-answer") {
    endConversation(callSid);
  }

  res.sendStatus(200);
});

export default router;

import db from "./database.js";
import config from "../config.js";

const { practice } = config;

/**
 * Generate all 30-minute slot start times for a given date
 * based on practice operating hours.
 */
function slotsForDate(dateStr) {
  const date = new Date(dateStr + "T00:00:00");
  const dayOfWeek = date.getDay();
  const hours = practice.hours[dayOfWeek];
  if (!hours) return []; // closed that day

  const slots = [];
  const [openH, openM] = hours.open.split(":").map(Number);
  const [closeH, closeM] = hours.close.split(":").map(Number);
  const openMinutes = openH * 60 + openM;
  const closeMinutes = closeH * 60 + closeM;

  for (let m = openMinutes; m + practice.slotDurationMinutes <= closeMinutes; m += practice.slotDurationMinutes) {
    const hh = String(Math.floor(m / 60)).padStart(2, "0");
    const mm = String(m % 60).padStart(2, "0");
    slots.push(`${hh}:${mm}`);
  }
  return slots;
}

/**
 * Return available time slots for a given date.
 */
export function getAvailableSlots(dateStr) {
  const allSlots = slotsForDate(dateStr);
  if (allSlots.length === 0) {
    const date = new Date(dateStr + "T00:00:00");
    const dayName = date.toLocaleDateString("en-US", { weekday: "long" });
    return { date: dateStr, closed: true, dayName, slots: [] };
  }

  const booked = db
    .prepare("SELECT time FROM appointments WHERE date = ?")
    .all(dateStr)
    .map((r) => r.time);

  const bookedSet = new Set(booked);
  const available = allSlots.filter((t) => !bookedSet.has(t));

  return { date: dateStr, closed: false, slots: available };
}

/**
 * Book an appointment. Returns the created appointment or an error.
 */
export function bookAppointment({ patientName, patientPhone, service, date, time, notes }) {
  // Validate service
  const matchedService = practice.services.find(
    (s) => s.toLowerCase() === service.toLowerCase()
  );
  if (!matchedService) {
    return { success: false, error: `Unknown service "${service}". Available: ${practice.services.join(", ")}` };
  }

  // Check the slot is actually available
  const { closed, slots } = getAvailableSlots(date);
  if (closed) {
    return { success: false, error: `The practice is closed on that day.` };
  }
  if (!slots.includes(time)) {
    return { success: false, error: `The ${time} slot on ${date} is not available.` };
  }

  const result = db
    .prepare(
      `INSERT INTO appointments (patient_name, patient_phone, service, date, time, duration_min, notes)
       VALUES (?, ?, ?, ?, ?, ?, ?)`
    )
    .run(patientName, patientPhone, matchedService, date, time, practice.slotDurationMinutes, notes || null);

  return {
    success: true,
    appointment: {
      id: result.lastInsertRowid,
      patientName,
      patientPhone,
      service: matchedService,
      date,
      time,
      durationMinutes: practice.slotDurationMinutes,
    },
  };
}

import Anthropic from "@anthropic-ai/sdk";
import config from "../config.js";
import { getAvailableSlots, bookAppointment } from "./appointments.js";

const anthropic = new Anthropic({ apiKey: config.anthropic.apiKey });

const { practice } = config;

// Build a human-readable hours string for the system prompt
const hoursDescription = Object.entries(practice.hours)
  .map(([day, hours]) => {
    const dayName = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"][day];
    return hours ? `${dayName}: ${hours.open} - ${hours.close}` : `${dayName}: Closed`;
  })
  .join("\n");

function buildSystemPrompt() {
  const today = new Date().toISOString().split("T")[0];
  return `You are a friendly, professional phone receptionist for ${practice.name}, a dental practice located at ${practice.address}. Your phone number is ${practice.phone}.

Your job is to help callers book dental appointments. Be warm, concise, and conversational — remember this is a phone call, not a chat. Keep responses short (1-3 sentences) since the caller is listening, not reading.

Practice hours:
${hoursDescription}

Available services: ${practice.services.join(", ")}

Today's date is ${today}.

To book an appointment you need to collect:
1. Patient's full name
2. A callback phone number (you can offer to use the number they're calling from)
3. The type of service / reason for visit
4. Preferred date and time

Workflow:
- Greet the caller and ask how you can help.
- If they want to book, collect the required info one or two pieces at a time. Don't ask for everything at once.
- Use the check_availability tool to find open slots before confirming a time.
- Use the book_appointment tool to finalize the booking.
- After booking, confirm the details back to the caller.
- If they have questions you can't answer (billing, specific treatment details), let them know you'll have someone from the office call them back.

Important guidelines:
- Speak naturally. Use phrases like "Sure thing!", "Let me check that for you", "Great, I've got you down for...".
- When reading times, say them naturally: "2:30 in the afternoon" not "14:30".
- When reading dates, say them naturally: "next Tuesday, March 5th" not "2026-03-05".
- If a date is fully booked, proactively suggest the next available day.
- Don't say "tool" or "function" or anything technical — just say "let me check".`;
}

const TOOLS = [

    name: "check_availability",
    description:

    input_schema: {
      type: "object",
      properties: {

          type: "string",
          description: "The date to check in YYYY-MM-DD format.",
        },
      },
      required: ["date"],
    },
  },
  {
    name: "book_appointment",
    description: "Book a dental appointment for a patient.",
    input_schema: {
      type: "object",
      properties: {
        patientName: {
          type: "string",
          description: "Full name of the patient.",
        },
        patientPhone: {
          type: "string",
          description: "Patient callback phone number.",
        },
        service: {
          type: "string",
          description: "Type of dental service (e.g. Cleaning, Exam, Filling).",
        },
        date: {
          type: "string",
          description: "Appointment date in YYYY-MM-DD format.",
        },
        time: {
          type: "string",
          description: "Appointment time in HH:MM 24-hour format.",
        },
        notes: {
          type: "string",
          description: "Any additional notes about the appointment.",
        },
      },
      required: ["patientName", "patientPhone", "service", "date", "time"],
    },
  },
];

/**
 * Execute a tool call from Claude and return the result.
 */
function executeTool(name, input) {
  switch (name) {
    case "check_availability":
      return getAvailableSlots(input.date);
    case "book_appointment":
      return bookAppointment(input);
    default:
      return { error: `Unknown tool: ${name}` };
  }
}

// In-memory conversation state per call.
// Map<CallSid, { messages: Array }>
const conversations = new Map();

/**
 * Process a caller's speech and return the assistant's text response.
 * Manages the full conversation loop including tool use.
 */
export async function chat(callSid, callerInput, callerPhone) {
  // Initialize conversation if new
  if (!conversations.has(callSid)) {
    conversations.set(callSid, { messages: [] });
  }

  const convo = conversations.get(callSid);

  // Add the caller's message
  if (callerInput) {
    convo.messages.push({ role: "user", content: callerInput });
  } else {
    // First turn — caller just picked up. Prompt the assistant to greet.
    convo.messages.push({
      role: "user",
      content: `[The caller just connected. Their phone number is ${callerPhone}. Greet them warmly.]`,
    });
  }

  // Loop to handle tool use — Claude may call tools and then respond.
  while (true) {
    const response = await anthropic.messages.create({
      model: "claude-sonnet-4-20250514",
      max_tokens: 300,
      system: buildSystemPrompt(),
      tools: TOOLS,
      messages: convo.messages,
    });

    // Add assistant response to history
    convo.messages.push({ role: "assistant", content: response.content });

    // Check if Claude wants to use tools
    const toolUseBlocks = response.content.filter((b) => b.type === "tool_use");

    if (toolUseBlocks.length === 0) {
      // No tool calls — extract the text response and return it.
      const text = response.content
        .filter((b) => b.type === "text")
        .map((b) => b.text)
        .join(" ");
      return text;
    }

    // Execute each tool call and feed results back
    const toolResults = toolUseBlocks.map((block) => ({
      type: "tool_result",
      tool_use_id: block.id,
      content: JSON.stringify(executeTool(block.name, block.input)),
    }));

    convo.messages.push({ role: "user", content: toolResults });
    // Continue the loop so Claude can respond with text after seeing tool results
  }
}

/**
 * Clean up conversation state when a call ends.
 */
export function endConversation(callSid) {
  conversations.delete(callSid);
}
